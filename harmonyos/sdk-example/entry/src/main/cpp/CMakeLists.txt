cmake_minimum_required(VERSION 3.20)
project(PicoInProcessSdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(PICO_SDK_EXAMPLE_SOURCE "$ENV{PICO_SDK_EXAMPLE_SOURCE}" CACHE PATH "Sibling SDK sources")
if(NOT PICO_SDK_EXAMPLE_SOURCE)
  get_filename_component(PICO_SDK_EXAMPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../sdk" ABSOLUTE)
endif()
if(NOT EXISTS "${PICO_SDK_EXAMPLE_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "SDK example source is required")
endif()
add_subdirectory("${PICO_SDK_EXAMPLE_SOURCE}" "${CMAKE_BINARY_DIR}/sdk-example")
add_library(picosdk SHARED sdk_napi.cpp)
target_compile_options(picosdk PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
target_link_libraries(picosdk PRIVATE pico-sdk-example libace_napi.z.so)
